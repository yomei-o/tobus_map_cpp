cmake_minimum_required(VERSION 3.16)
project(tobus_map CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /O2 /wd4244 /wd4267 /wd4996)
endif()
add_executable(tobus_map main.cpp)
target_include_directories(tobus_map PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} third_party third_party/nlohmann)
